def num(x):
    try: return float(x) if x not in ('', None, 'None') else None
    except ValueError: return None

def parse(path):
    out = {}
    for line in open(path).read().strip().split('\n'):
        if not line.strip(): continue
        p = line.split('|'); out.setdefault(p[0], []).append(p[1:])
    return out

def pct(a, b): return None if (a is None or b in (None, 0)) else (a / b - 1) * 100
def med(xs):
    xs = [x for x in xs if x is not None]
    return st.median(xs) if xs else None
def sgn(x, d=1): return ('+' if x > 0 else '') + f'{x:.{d}f}' if x is not None else 'n/a'

def _gnum(x):
    try: return float(x)
    except (TypeError, ValueError): return None

def commodity_indicators(S, R, px, put, wk_closes):
    """Score the G- indicators from the collector lines (R = rawA + rawG lines merged). Returns the facts shown on the tab."""
    spec = {i['id']: i for i in (S.get('commodity') or {}).get('indicators', [])}
    fred = {r[0]: r for r in R.get('FRED', [])}; sp = (R.get('SPDR') or [None])[0]; cf = (R.get('CFTC') or [None])[0]; ra = (R.get('RATIO') or [None])[0]
    MON_ = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec']
    dl = lambda iso: f'{int(iso[8:10])} {MON_[int(iso[5:7]) - 1]} {iso[:4]}' if iso else '?'
    def score(id_, v):
        if v is None: return None
        kind, bull, bear = G_RULES[id_]
        if kind == 'hi': return 1 if v >= bull else -1 if v <= bear else 0
        if kind == 'lo<': return 1 if v < bull else -1 if v >= bear else 0
        return 1 if v <= bull else -1 if v >= bear else 0
    def g(id_, v, text, note):
        if id_ not in spec: return
        c = spec[id_]
        put(id_, G_BUCKET.get(c['bucket'], c['bucket']), c['name'], c['weight'], v, text if v is not None else 'n/a', score(id_, v),
            c['measured_against'], G_WORDS[id_][0], G_WORDS[id_][1], G_TIMING[id_], G_SOURCE.get(id_, c.get('source', '')), note)
    facts = {}
    # fund holdings (SPDR daily file): SPDR|date|tonnes|prevdate|prev|d20|t20|d60|t60|premium%|oz/share|value$
    if sp:
        t0, t20, t60 = _gnum(sp[1]), _gnum(sp[5]), _gnum(sp[7])
        v = pct(t0, t20); g('G-LQ2', v, f'{sgn(v,1)}%', f'{t0:,.1f} tonnes on {dl(sp[0])} vs {t20:,.1f} tonnes on {dl(sp[4])}.' if v is not None else '')
        v = pct(t0, t60); g('G-DM3', v, f'{sgn(v,1)}%', f'{t0:,.1f} tonnes on {dl(sp[0])} vs {t60:,.1f} tonnes on {dl(sp[6])}. GLD only, not all gold funds worldwide.' if v is not None else '')
        facts['spdr'] = {'date': sp[0], 'tonnes': t0, 'prev_date': sp[2], 'prev': _gnum(sp[3]), 'd20': sp[4], 't20': t20, 'd60': sp[6], 't60': t60,
                         'premium_pct': _gnum(sp[8]), 'oz_per_share': _gnum(sp[9]), 'value': _gnum(sp[10]) if len(sp) > 10 else None}
        if len(sp) > 11 and sp[11]: facts['spdr']['series'] = {'start': sp[11], 'v': [int(x) / 10 for x in sp[12].split(',') if x]} if len(sp) > 12 else None
    else:
        g('G-LQ2', None, 'n/a', 'SPDR daily file not collected today.'); g('G-DM3', None, 'n/a', 'SPDR daily file not collected today.')
    # positioning (CFTC): CFTC|report|net%OI|pctile|prevdate|prev|4wkdate|4wk|max3y|min3y
    if cf:
        nl, pc_ = _gnum(cf[1]), _gnum(cf[2])
        g('G-LQ3', pc_, f'{pc_:.0f}th percentile' if pc_ is not None else 'n/a',
          f"Read the other way round: crowded = bearish. Hedge funds hold a net {nl:.1f}% of all gold futures (report of {dl(cf[0])}); 3-year range {_gnum(cf[8]):.1f}% to {_gnum(cf[7]):.1f}%." if nl is not None else '')
        facts['cftc'] = {'report': cf[0], 'net_long_pct_oi': nl, 'pctile_3y': pc_, 'prev_date': cf[3], 'prev': _gnum(cf[4]), 'd4w': cf[5], 'v4w': _gnum(cf[6]), 'max3y': _gnum(cf[7]), 'min3y': _gnum(cf[8])}
    else:
        g('G-LQ3', None, 'n/a', 'CFTC report not collected today.')
    # rates (FRED): FRED|series|date|value|date20|value20|prevdate|prev
    for id_, ser in (('G-DM1', 'DFII10'), ('G-DM2', 'T10YIE')):
        r = fred.get(ser)
        if r and _gnum(r[2]) is not None and _gnum(r[4]) is not None:
            v = round(_gnum(r[2]) - _gnum(r[4]), 2)
            g(id_, v, f'{sgn(v,2)} pts', f'{_gnum(r[2]):.2f}% on {dl(r[1])} vs {_gnum(r[4]):.2f}% on {dl(r[3])}.' + (' FRED publishes a day late.' if ser == 'DFII10' else ''))
            facts.setdefault('fred', {})[ser] = {'d': r[1], 'v': _gnum(r[2]), 'd20': r[3], 'v20': _gnum(r[4])}
        else:
            g(id_, None, 'n/a', f'FRED {ser} not collected today.')
    # stretch: price vs 200-week average (weekly closes incl. the current week)
    if wk_closes and len(wk_closes) >= 200:
        w200 = sum(wk_closes[-200:]) / 200; v = pct(px, w200)
        g('G-VL1', v, f'{sgn(v)}%', f'${px:.2f} vs a 200-week average of ${w200:.2f}.'); facts['w200'] = round(w200, 2)
    else:
        g('G-VL1', None, 'n/a', f'needs 200 weekly closes ({len(wk_closes or [])} collected).')
    # stretch: gold ÷ S&P 500 vs its 10-year median: RATIO|week|gold|spx|ratio|median|from|to|n
    if ra:
        rr, med = _gnum(ra[3]), _gnum(ra[4]); v = pct(rr, med)
        g('G-VL2', v, f'{sgn(v)}%', f'Ratio {rr:.3f} (gold ${_gnum(ra[1]):,.0f} ÷ S&P 500 {_gnum(ra[2]):,.0f}, week of {dl(ra[0])}) vs a 10-year median of {med:.3f} ({ra[7]} weeks).')
        facts['ratio'] = {'week': ra[0], 'gold': _gnum(ra[1]), 'spx': _gnum(ra[2]), 'ratio': rr, 'median10y': med, 'from': ra[5], 'to': ra[6], 'n': ra[7]}
    else:
        g('G-VL2', None, 'n/a', 'Gold ÷ S&P 500 history not collected today.')
    return facts

SCORE_WORD = {1: 'Bullish', 0: 'Neutral', -1: 'Bearish', None: 'No data'}

def explain(it, S):
    k = it['id']
    if k in EXPLAIN: what, why, s = EXPLAIN[k]
    else:
        cfg = next((d for d in (S.get('dm4', []) + ([S['dm5']] if S.get('dm5') else [])) if d['id'] == k), {})
        what = cfg.get('what', it['name']); why = cfg.get('why', '')
        s = {1: cfg.get('bull_means', 'Better than the bullish line.'), 0: cfg.get('neutral_means', 'Between the lines.'), -1: cfg.get('bear_means', 'Worse than the bearish line.')}
    if S.get('asset_type') == 'commodity' and k in EXPLAIN_COMMOD:
        cm = S.get('commodity') or {}; asset = cm.get('asset', S.get('name', 'it')); peer = cm.get('pt4_label', ', '.join(p['name'] for p in S.get('peers', [])))
        fm = lambda x: x.format(asset=asset, asset_cap=asset[:1].upper() + asset[1:], peer=peer)
        w2, y2, s2 = EXPLAIN_COMMOD[k]
        what = fm(w2) if w2 else what; why = fm(y2) if y2 else why; s = {kk: fm(vv) for kk, vv in s2.items()}
    if k in ('PT1', 'PT2') and 'since listing' in (it.get('measured') or ''):   # new-listing mode wording
        what = ('Whether the share price is above or below its average price since listing (the long-run average does not exist yet).' if k == 'PT1'
                else 'Whether the last 20 trading days are running above or below the average since listing (short-history version of the trend check).')
    return {'what': what, 'why': why, 'score': s.get(it['s'], 'No live data today, so it is left out of the score.')}

# ---------------------------------------------------------------- holder display names
ALIAS = {'fmr': 'Fidelity (FMR)', 'price t rowe associates': 'T. Rowe Price', 'blackrock': 'BlackRock', 'jpmorgan chase': 'JPMorgan Chase',
         'geode capital management': 'Geode Capital', 'state street': 'State Street', 'ubs am': 'UBS Asset Management'}
def clean_name(n):
    import re
    s = re.sub(r'/[A-Za-z]{2}/', '', n)
    s = re.sub(r'\b(Inc|Llc|Lp|Llp|Corp|Corporation|Ltd|Plc|Co|Group Inc)\b\.?', '', s, flags=re.I)
    s = re.sub(r'[\s,\.&]+$', '', re.sub(r'\s+', ' ', s)).strip(' ,.')
    low = s.lower()
    for k, v in ALIAS.items():
        if low.startswith(k): return v
    return s

# ---------------------------------------------------------------- overnight changes (what moved since the previous refresh)
# Materiality lines, each measured against the previous refresh:
#   composite score moves 10+ points; a bucket moves 25+ points; any indicator score changes; the call or zone changes;
#   peer premium on forward P/E moves 10+ percentage points; a chain status changes or its watch number crosses its line;
#   a results date is added, confirmed or moved, or new results arrive; a holder's classification or the filing quarter changes;
#   the quality flag count changes; an unusual price move (2x a typical day); a new flag or override.
def changes(prev, d):
    out = {k: [] for k in ('call', 'changers', 'move', 'calendar', 'indicators', 'valuation', 'mix', 'earnings', 'shareholders', 'chains', 'quality', 'price')}
    if not prev or prev.get('meta', {}).get('asof') == d['meta']['asof']:
        return {'since': None, 'first': not prev, 'sections': out, 'count': 0}
    since = prev['meta']['asof']
    pc, nc = prev.get('composite'), d.get('composite')
    if pc is not None and nc is not None and abs(nc - pc) >= 10: out['call'].append(f'Composite score {pc:+d} → {nc:+d}')
    if (prev.get('call') or {}).get('code') != (d.get('call') or {}).get('code') and d['call'].get('code'):
        out['call'].append(f"Call changed: {prev['call'].get('label','none')} → {d['call']['label']}")
    elif prev.get('zone') != d.get('zone'):
        out['call'].append(f"Today's zone moved: {prev.get('zone')} → {d.get('zone')} (not yet confirmed)")
    for b, v in d['buckets'].items():
        pv = (prev.get('buckets') or {}).get(b)
        if pv is not None and v is not None and abs(v - pv) >= 25: out['call'].append(f'{b} bucket {pv:+d} → {v:+d}')
    pf = {f['code'] for f in prev.get('flags', [])}
    for f in d['flags']:
        if f['code'] not in pf: out['call'].append('New flag: ' + f['text'])
    pi = {i['id']: i for i in prev.get('indicators', [])}
    for i in d['indicators']:
        p = pi.get(i['id'])
        if p and p.get('s') != i['s']:
            out['indicators'].append({'id': i['id'], 'name': i['name'], 'from': SCORE_WORD[p.get('s')], 'to': SCORE_WORD[i['s']], 'w': i['w'],
                                      'up': (i['s'] or 0) > (p.get('s') or 0)})
            # counted once, under Indicators (the valuation and 'what would change' sections no longer repeat it)
    def prem(x): m = x['peers']['median'].get('fpe'); s = x['peers']['stock'].get('fpe'); return (s / m - 1) * 100 if s and m else None
    a, b = prem(prev), prem(d)
    if a is not None and b is not None and abs(b - a) >= 10: out['valuation'].append(f'Forward P/E premium to peers {a:+.0f}% → {b:+.0f}%')
    if d['move'].get('unusual'): out['move'].append(f"Unusual move: {d['move']['d1']*100:+.1f}% ({d['move']['ratio']:.1f}x a typical day); classified {d['move'].get('class','n/a')}")
    elif prev.get('move', {}).get('class') != d['move'].get('class'): out['move'].append(f"Classification {prev['move'].get('class')} → {d['move'].get('class')}")
    pcal = {c['t'] for c in prev.get('calendar', [])}
    for c in d['calendar']:
        if c['t'] not in pcal: out['calendar'].append('New: ' + c['t'] + ' (' + c['d'] + ')')
    pn = {n['q']: n for n in prev.get('earnings', {}).get('next', [])}
    for n in d['earnings'].get('next', []):
        p = pn.get(n['q'])
        if not p: out['earnings'].append(f"Next results added: {n['q']}")
        elif p.get('date') != n.get('date') or p.get('status') != n.get('status'): out['earnings'].append(f"{n['q']}: {p.get('date')} ({p.get('status')}) → {n['date']} ({n['status']})")
    pq = [e['q'] for e in prev.get('earnings', {}).get('past', [])]
    for e in d['earnings'].get('past', []):
        if e['q'] not in pq: out['earnings'].append(f"New results: {e['q']}")
    if prev.get('shareholders', {}).get('asof') != d['shareholders'].get('asof'): out['shareholders'].append(f"New filing quarter: {d['shareholders'].get('asof')}")
    else:
        pcls = {h['name']: h['cls'] for h in prev['shareholders'].get('top5', []) + prev['shareholders'].get('active', [])}
        for h in d['shareholders']['top5'] + d['shareholders']['active']:
            if h['name'] in pcls and pcls[h['name']] != h['cls']: out['shareholders'].append(f"{h['disp']}: {pcls[h['name']]} → {h['cls']}")
    pch = {c['id']: c for c in prev.get('chains', [])}
    for c in d['chains']:
        p = pch.get(c['id'])
        if p and p.get('status') != c.get('status'): out['chains'].append(f"{c['id']} {c['name']}: {p['status']} → {c['status']}")
    out['mix'] = mix_changes(prev.get('mix'), d.get('mix'))
    if prev.get('quality', {}).get('flags') != d['quality']['flags']: out['quality'].append(f"Red flags {prev['quality']['flags']} → {d['quality']['flags']}")
    pc_, nc_ = (prev.get('quality', {}).get('cash') or {}), (d['quality'].get('cash') or {})
    if nc_.get('latest') and pc_.get('latest') and nc_['latest'] != pc_['latest'] and nc_.get('ttm') is not None:
        out['quality'].append(f"New results: free cash flow {'−' if nc_['ttm'] < 0 else ''}${abs(nc_['ttm'])/1e3:.1f}bn ({nc_['ttm_label']})")
    if d['move'].get('unusual'): out['price'].append(f"{d['move']['d1']*100:+.1f}% on the day")
    n = sum(len(v) for v in out.values())
    return {'since': since, 'first': False, 'sections': out, 'count': n}

# ---------------------------------------------------------------- revenue mix (plan v0.4, section "Revenue mix"; NOT scored)
# Input : stage D bundle (SEC inline XBRL, US) or side.json mix.quarters (manual, any market) + settings.mix + side.json mix text.
# Output: data['mix'] = quarters for the chart, one row per segment (shares, growth, margin), recurring share, mix effect on margin,
#         automatic plus/minus signals, and the chat-written "what it means" text.
# Windows (all stated on the page): "now" = last 4 reported quarters; "1 year ago" = the 4 quarters before that;
#         "3 years ago" = the 4 quarters ending 12 quarters before the latest (or the earliest full 4 quarters held, labelled).

MIX_MOVE = 2.0        # change flag: a segment's last-4-quarter share moves 2+ percentage points vs the previous refresh
MIX_OTHER = 0.005     # add an "Other" slice when reported segments miss the total by more than 0.5% of revenue
MIX_MAX = 6           # validated palette has 6 colours; smaller segments beyond that fold into "Other"
MIX_EDGE = 3.0        # a segment/group margin counts as "higher" or "lower" margin when 3+ points away from the company figure

def _d(s): return _dt.date.fromisoformat(s)
def _qlabel(e):
    d = _d(e)
    if d.day <= 7: d = d - _dt.timedelta(days=d.day)   # a quarter ending 1 Apr is the March quarter
    return d.strftime('%b %Y')   # quarter named by the month it ends (fiscal quarters differ by company)

def _mwords(m):
    return re.sub(r'(?<=[a-z])(?=[A-Z])', ' ', m).lower().strip()

def _series(D, concept):
    """Groups the stage D lines of one concept into series keyed by the printed row label (filers' tags can be wrong).
    If keys sharing a label disagree on a quarter (a generic label such as "cost of" used by every segment), that label is
    split by member instead. Each series: {'Q': {end: v}, 'FY': {end: v}, 'mw': member words}. Quarter 4 = year - 3 quarters."""
    E = D['SEGQ'][0][0].split(',') if 'SEGQ' in D else []
    keys = {}
    for row in D.get('SEG', []):
        if row[0] != concept: continue
        k = keys.setdefault(row[1], {'Q': {}, 'FY': {}})
        for e, v in zip(E, row[2].split(',')):
            if v != '' and e not in k['Q']: k['Q'][e] = float(v)
    for row in D.get('SEGFY', []):
        if row[0] != concept: continue
        k = keys.setdefault(row[1], {'Q': {}, 'FY': {}})
        for pair in row[2].split(' '):
            if ':' in pair:
                e, v = pair.split(':'); k['FY'].setdefault(e, float(v))
    bylab = {}
    for key, k in keys.items():
        lab = key.split('~', 1)[1] if '~' in key else _mwords(key)
        bylab.setdefault(lab, []).append((key, k))
    out = {}
    for lab, items in bylab.items():
        clash = False
        for kind in ('Q', 'FY'):
            seen = {}
            for key, k in items:
                for e, v in k[kind].items():
                    if e in seen and abs(seen[e] - v) > 1: clash = True
                    seen.setdefault(e, v)
        groups = {lab: items}
        if clash:
            groups = {}
            for key, k in items: groups.setdefault(_mwords(key.split('~')[0]), []).append((key, k))
        for name, its in groups.items():
            s = out.setdefault(name, {'Q': {}, 'FY': {}, 'mw': set()})
            for key, k in its:
                s['mw'].add(_mwords(key.split('~')[0]))
                for e, v in k['Q'].items(): s['Q'].setdefault(e, v)
                for e, v in k['FY'].items(): s['FY'].setdefault(e, v)
    for s in out.values():
        for e, fy in s['FY'].items():
            if e in s['Q']: continue
            ins = [q for q in E if _d(e) - _dt.timedelta(days=300) < _d(q) < _d(e)]
            if len(ins) == 3 and all(q in s['Q'] for q in ins): s['Q'][e] = fy - sum(s['Q'][q] for q in ins)
    return E, out

def _pick(series, patterns, e):
    """First pattern that gives a complete value for quarter e; a pattern matching several series adds them up.
    Patterns are tested on the printed label; if no label matches, on the member (tag) words."""
    for p in patterns:
        rx = re.compile(p, re.I); labs = [l for l in series if rx.search(l)]
        if not labs: labs = [l for l in series if any(rx.search(w) for w in series[l]['mw'])]
        if not labs: continue
        vals = [series[l]['Q'].get(e) for l in labs]
        if all(v is not None for v in vals): return sum(vals)
    return None

SNAP_DAYS = 40   # trading days of day-by-day detail kept for the timeline (plan S16)
SNAP_KEYS = ('price', 'call', 'zone', 'composite', 'coverage', 'buckets', 'bucket_live', 'driver_notes', 'direction', 'confidence',
             'flags', 'changers', 'lead', 'bullets', 'move', 'changes', 'new_listing')

def snap_of(d):
    """One day's view for the timeline: the call, scores, headline words, what's happening and that day's change flags."""
    s = {k: d.get(k) for k in SNAP_KEYS}
    s['built'] = (d.get('meta') or {}).get('built')   # when that day's refresh ran (UTC), shown as 'updated' in SGT
    s['indicators'] = [{'id': i['id'], 's': i.get('s'), 'v': i.get('v'), 'text': i.get('text'), 'note': i.get('note'),
                        'score': (i.get('explain') or {}).get('score')} for i in d.get('indicators', [])]
    return s

def snapshots(prev, d):
    """Carry the previous days' snapshots forward, add today's, keep the last SNAP_DAYS. A previous data.json written before
    snapshots existed seeds its own day from itself, so no recorded day is lost."""
    asof = d['meta']['asof']; out = {}
    if prev:
        out.update({k: v for k, v in (prev.get('snapshots') or {}).items() if k < asof})
        pa = (prev.get('meta') or {}).get('asof')
        if pa and pa < asof and pa not in out and prev.get('indicators'): out[pa] = snap_of(prev)
    out[asof] = snap_of(d)
    return {k: out[k] for k in sorted(out)[-SNAP_DAYS:]}

def build_mix(S, D, side, prev_mix=None, currency='USD'):
    cfg = S.get('mix'); txt = side.get('mix', {}) if side else {}
    if not cfg: return None
    basis = cfg.get('basis', 'gross')
    segs_cfg = cfg['segments']; groups_cfg = cfg.get('groups') or [{'id': s['id'], 'name': s['name'], 'match': s['match']} for s in segs_cfg]
    # ---- quarterly numbers per segment / group
    Q = []   # list of {end, total, rev{id}, grev{gid}, gprof{gid}}
    if cfg.get('source', 'sec') == 'manual':
        for q in txt.get('quarters', []):
            Q.append({'end': q['end'], 'total': q.get('total') or sum(v for v in q['rev'].values() if v is not None), 'rev': q['rev'],
                      'grev': q.get('group_rev') or {g['id']: sum(q['rev'].get(s['id']) or 0 for s in segs_cfg if s.get('group', s['id']) == g['id']) for g in groups_cfg},
                      'gprof': q.get('profit', {})})
        Q.sort(key=lambda x: x['end']); src = txt.get('source', 'Company results (entered by the stock chat)')
    else:
        if not D or 'SEGQ' not in D:
            # stage D is only run when a new filing may have landed; otherwise carry yesterday's mix unchanged (production rule, 30 Sep 2026)
            if prev_mix and not prev_mix.get('error'): return dict(prev_mix, carried=True)
            return {'error': 'No segment data (stage D not run)'}
        E, REV = _series(D, 'REV'); _, COST = _series(D, 'COST'); _, OI = _series(D, 'OI'); _, GP = _series(D, 'GP')
        for e in E:
            tot = _pick(REV, ['^total$'], e)
            rev = {s['id']: _pick(REV, s['match'], e) for s in segs_cfg}
            grev = {g['id']: _pick(REV, g['match'], e) for g in groups_cfg}
            if basis == 'operating':
                gprof = {g['id']: _pick(OI, g['match'], e) for g in groups_cfg}
            else:
                gprof = {}
                for g in groups_cfg:
                    c = _pick(COST, g['match'], e); gprof[g['id']] = grev[g['id']] - c if (c is not None and grev[g['id']] is not None) else _pick(GP, g['match'], e)
            Q.append({'end': e, 'total': tot, 'rev': rev, 'grev': grev, 'gprof': gprof,
                      'co_gp': _pick(GP, ['^total$'], e), 'co_oi': _pick(OI, ['^total$'], e)})
        s0 = D.get('SEGSRC', [['', '', '']])[0]
        src = f"SEC filings (10-Q / 10-K inline XBRL), {s0[0]}; quarter 4 = full year minus the first three quarters"
    # ---- keep the latest unbroken run of quarters that have a total (and at least one segment)
    run = []
    for q in reversed(Q):
        if q['total'] is None or all(v is None for v in q['rev'].values()):
            if run: break
            continue
        if run and (_d(run[-1]['end']) - _d(q['end'])).days > 100: break
        run.append(q)
    run.reverse(); run = run[-16:]
    # short history (fewer than 8 unbroken quarters, e.g. a new listing): show every quarter held (gaps allowed) and compare
    # single quarters (latest vs the same quarter a year earlier) instead of 4-quarter sums
    short = len(run) < 8
    if short:
        run = [q for q in Q if q['total'] is not None and any(v is not None for v in q['rev'].values())][-16:]
    if len(run) < 2: return {'error': 'Fewer than 2 quarters of segment data'}
    # ---- "Other" and folding beyond 6 colours
    segs = [dict(s) for s in segs_cfg]
    gap = max(abs(q['total'] - sum(v or 0 for v in q['rev'].values())) / q['total'] for q in run if all(v is not None for v in q['rev'].values())) if any(all(v is not None for v in q['rev'].values()) for q in run) else 0
    fold = []
    if len(segs) > MIX_MAX:
        size = {s['id']: sum(q['rev'].get(s['id']) or 0 for q in run[-4:]) for s in segs}
        keep = sorted(segs, key=lambda s: -size[s['id']])[:MIX_MAX - 1]; fold = [s for s in segs if s not in keep]; segs = [s for s in segs if s in keep]
    if gap > MIX_OTHER or fold:
        for q in run:
            known = sum(q['rev'].get(s['id']) or 0 for s in segs)
            q['rev']['OTHER'] = q['total'] - known if all(q['rev'].get(s['id']) is not None for s in segs) else None
        segs.append({'id': 'OTHER', 'name': 'Other' + (' (' + ', '.join(s['name'] for s in fold) + ')' if fold else ''), 'recurring': None, 'group': None, 'auto': True})
    ids = [s['id'] for s in segs]
    # ---- windows
    L = len(run) - 1
    def win(k):   # 4 quarters ending k quarters before the latest; None if not held
        a = L - k - 3
        return run[a:L - k + 1] if a >= 0 else None
    if short:
        def ago(days):
            c = [q for q in run[:-1] if abs((_d(run[-1]['end']) - _d(q['end'])).days - days) <= 20]
            return [c[-1]] if c else None
        W = {'now': [run[-1]], 'y1': ago(364), 'y3': ago(1092)}
        lab = {k: ('quarter to ' + _qlabel(w[-1]['end']) if w else None) for k, w in W.items()}
    else:
        W = {'now': win(0), 'y1': win(4), 'y3': win(12)}
        if W['y3'] is None and len(run) >= 12:   # earliest full 4 quarters held, labelled
            W['y3'] = run[0:4]
        lab = {k: ('4 quarters to ' + _qlabel(w[-1]['end']) if w else None) for k, w in W.items()}
    def tsum(w, f):
        if not w: return None
        vals = [f(q) for q in w]; return sum(vals) if all(v is not None for v in vals) else None
    tot = {k: tsum(w, lambda q: q['total']) for k, w in W.items()}
    def share(sid, k):
        r = tsum(W[k], lambda q: q['rev'].get(sid)); return r / tot[k] * 100 if (r is not None and tot[k]) else None
    def growth(sid):
        a = tsum(W['now'], lambda q: q['rev'].get(sid)); b = tsum(W['y1'], lambda q: q['rev'].get(sid))
        return (a / b - 1) * 100 if (a is not None and b) else None
    def qgrowth(sid):   # latest quarter vs the same quarter a year earlier
        if short: return growth(sid)
        if L < 4: return None
        a, b = run[L]['rev'].get(sid), run[L - 4]['rev'].get(sid); return (a / b - 1) * 100 if (a is not None and b) else None
    # ---- margins per group (last 4 quarters)
    groups = []
    for g in groups_cfg:
        gm = {}
        for k in ('now', 'y1', 'y3'):
            r = tsum(W[k], lambda q: q['grev'].get(g['id'])); p = tsum(W[k], lambda q: q['gprof'].get(g['id']))
            gm[k] = p / r * 100 if (r and p is not None) else None
        gs = {k: tsum(W[k], lambda q: q['grev'].get(g['id'])) for k in ('now', 'y1', 'y3')}
        groups.append({'id': g['id'], 'name': g['name'], 'margin': gm, 'rev': gs, 'segments': [s['id'] for s in segs_cfg if s.get('group', s['id']) == g['id']]})
    # company figure on the same basis = all groups together (before costs the company does not allocate)
    co = {}
    for k in ('now', 'y1', 'y3'):
        rs = [x['rev'][k] for x in groups]; ps = [tsum(W[k], lambda q, gid=x['id']: q['gprof'].get(gid)) for x in groups]
        co[k] = sum(ps) / sum(rs) * 100 if all(v is not None for v in rs + ps) and sum(rs) else None
    # mix effect: margin change from the shift in group weights alone, holding each group's margin at its year-earlier level
    def gw(k): return {x['id']: (x['rev'][k] / sum(y['rev'][k] for y in groups)) if all(y['rev'][k] for y in groups) else None for x in groups}
    def effect(k0):
        w1, w0 = gw('now'), gw(k0)
        if any(v is None for v in list(w1.values()) + list(w0.values())) or any(x['margin'][k0] is None for x in groups): return None
        return sum((w1[x['id']] - w0[x['id']]) * x['margin'][k0] for x in groups)
    mix_eff = {'y1': effect('y1'), 'y3': effect('y3') if W['y3'] else None}
    # ---- per segment rows
    gmap = {x['id']: x for x in groups}; tsegs = txt.get('segments', {})
    rows = []
    for i, s in enumerate(segs):
        g = gmap.get(s.get('group', s['id'])) if not s.get('auto') else None
        m_now = g['margin']['now'] if g else None
        tag = None
        if m_now is not None and co['now'] is not None:
            tag = 'higher' if m_now >= co['now'] + MIX_EDGE else 'lower' if m_now <= co['now'] - MIX_EDGE else 'similar'
        t = tsegs.get(s['id'], {})
        rows.append({'id': s['id'], 'name': s['name'], 'slot': i, 'recurring': s.get('recurring'), 'group': g['name'] if g else None,
                     'group_shared': bool(g and len(g['segments']) > 1),
                     'share': {k: share(s['id'], k) for k in ('now', 'y1', 'y3')}, 'growth': growth(s['id']), 'qgrowth': qgrowth(s['id']),
                     'margin': (g['margin'] if g else None), 'margin_tag': tag,
                     'sells': t.get('sells', ''), 'buyers': t.get('buyers', ''), 'pos': t.get('pos', []), 'neg': t.get('neg', [])})
    co_growth = (tot['now'] / tot['y1'] - 1) * 100 if (tot['now'] and tot['y1']) else None
    if short: co_qg = co_growth
    else: co_qg = (run[L]['total'] / run[L - 4]['total'] - 1) * 100 if L >= 4 and run[L - 4]['total'] else None
    # ---- recurring share and higher-margin share
    def rec_share(k, levels=('yes',)):
        v = [r['share'][k] for r in rows if r['recurring'] in levels]; return sum(v) if v and all(x is not None for x in v) else (0.0 if not v else None)
    recurring = {k: rec_share(k) for k in ('now', 'y1', 'y3')} if any(r['recurring'] == 'yes' for r in rows) else None
    partly = {k: rec_share(k, ('partly',)) for k in ('now', 'y1', 'y3')} if any(r['recurring'] == 'partly' for r in rows) else None
    # ---- automatic signals (facts only; the chat writes the judgement in side.json mix.shift)
    wtxt = 'latest quarter' if short else 'last 4 quarters'
    wcmp = (lambda l: f'latest quarter vs the same quarter {l} earlier') if short else (lambda l: f'last 4 quarters vs the 4 quarters {l} earlier')
    pos, neg = [], []
    basis_w = 'gross margin' if basis == 'gross' else 'segment operating margin'
    for k, lbl in (('y1', 'a year'), ('y3', '3 years')):
        e = mix_eff.get(k)
        if e is None: continue
        if e >= 0.3: pos.append(f"The shift in mix alone added {e:.1f} pts to {basis_w} vs {lbl} earlier ({wcmp(lbl)}, each part's margin held at that earlier level).")
        elif e <= -0.3: neg.append(f"The shift in mix alone took {abs(e):.1f} pts off {basis_w} vs {lbl} earlier ({wcmp(lbl)}, each part's margin held at that earlier level).")
    if recurring and recurring['now'] is not None and recurring['y1'] is not None:
        dv = recurring['now'] - recurring['y1']
        if dv >= 1: pos.append(f"Recurring revenue share up {dv:.1f} pts in a year to {recurring['now']:.0f}% of sales ({wtxt}).")
        elif dv <= -1: neg.append(f"Recurring revenue share down {abs(dv):.1f} pts in a year to {recurring['now']:.0f}% of sales ({wtxt}).")
    if partly and partly['now'] is not None and partly['y3'] is not None and abs(partly['now'] - partly['y3']) >= 1:
        dv = partly['now'] - partly['y3']
        (pos if dv > 0 else neg).append(f"Partly recurring revenue (long-life contracts or design wins) {'up' if dv > 0 else 'down'} {abs(dv):.0f} pts in 3 years to {partly['now']:.0f}% of sales ({wtxt}).")
    fastest = max((r for r in rows if r['growth'] is not None and not r['id'] == 'OTHER'), key=lambda r: r['growth'], default=None)
    if fastest and fastest['margin_tag'] and co_growth is not None and fastest['growth'] > co_growth:
        if fastest['margin_tag'] == 'higher': pos.append(f"Fastest-growing part ({fastest['name']}, {fastest['growth']:+.0f}% vs a year earlier) earns above the company {basis_w}.")
        elif fastest['margin_tag'] == 'lower': neg.append(f"Fastest-growing part ({fastest['name']}, {fastest['growth']:+.0f}% vs a year earlier) earns below the company {basis_w}.")
    top = max((r for r in rows if r['share']['now'] is not None), key=lambda r: r['share']['now'], default=None)
    if top and top['share']['now'] >= 50: neg.append(f"Concentration: {top['name']} is {top['share']['now']:.0f}% of sales ({wtxt}).")
    # ---- data checks
    checks = []
    if gap > MIX_OTHER: checks.append(f"Reported parts miss the total by up to {gap*100:.1f}% of revenue: shown as Other.")
    miss = [s['name'] for s in segs if any(q['rev'].get(s['id']) is None for q in run[-4:])]
    if miss: checks.append('Missing in a recent quarter: ' + ', '.join(miss) + '.')
    # rolling margin (last 4 quarters at each quarter end) per group and for all groups together: the margin chart (D20, 30 Sep)
    mser = {}
    span = 1 if short else 4   # short history: each point is one quarter
    def roll(i, gids):
        w = run[i - span + 1:i + 1]
        r = [q['grev'].get(g) for q in w for g in gids]; p = [q['gprof'].get(g) for q in w for g in gids]
        return round(sum(p) / sum(r) * 100, 2) if all(v is not None for v in r + p) and sum(r) else None
    for g in groups: mser[g['id']] = [roll(i, [g['id']]) for i in range(span - 1, len(run))]
    mser['_co'] = [roll(i, [g['id'] for g in groups]) for i in range(span - 1, len(run))]
    latest = run[-1]['end']
    out = {'unit': '$m' if currency == 'USD' else f'{currency} m', 'currency': currency, 'basis': basis, 'basis_word': basis_w, 'source': src,

           'quarters': [{'end': q['end'], 'label': _qlabel(q['end']), 'total': q['total'], 'rev': {i: q['rev'].get(i) for i in ids}} for q in run],
           'segments': rows, 'groups': [{'id': g['id'], 'name': g['name'], 'margin': g['margin'], 'segments': g['segments'],
                                         'slot': next((i for i, s in enumerate(segs) if s['id'] in g['segments']), 0), 'series': mser[g['id']]} for g in groups],
           'margin_series': {'ends': [run[i]['end'] for i in range(span - 1, len(run))], 'labels': [_qlabel(run[i]['end']) for i in range(span - 1, len(run))], 'company': mser['_co'], 'span': span},
           'short': short,

           'signals': {'pos': pos, 'neg': neg}, 'shift': txt.get('shift', {}), 'note': cfg.get('note', ''), 'checks': checks,
           'scored': False}
    return out

def mix_changes(prev_mix, mix):
    out = []
    if not mix or 'segments' not in mix: return out
    if not prev_mix or 'segments' not in prev_mix: return out
    if prev_mix.get('latest') != mix['latest']: out.append(f"New results: mix now runs to the {mix['latest_label']} quarter")
    ps = {r['id']: r['share']['now'] for r in prev_mix['segments']}
    for r in mix['segments']:
        a, b = ps.get(r['id']), r['share']['now']
        if a is not None and b is not None and abs(b - a) >= MIX_MOVE: out.append(f"{r['name']} share of sales {a:.1f}% → {b:.1f}% (last 4 quarters)")
    return out

# ---------------------------------------------------------------- cash flow and interest from SEC periods (plan v0.5, S10)
# 10-Q cash-flow statements are year-to-date: quarters are derived by difference (6 months - 3 months, 9 - 6, year - 9, and
# half-year - its second quarter). Stage B lines: CFP|start|end|operating cash|capex|filed and INTP|start|end|op income|interest|filed ($m).
def _days(a, b): return (_dt.date.fromisoformat(b) - _dt.date.fromisoformat(a)).days
def _qseries(rows, idx, asof):
    P = {}
    for r in rows:
        if len(r) <= idx or r[idx] in ('', None) or (len(r) > 4 and r[4] and r[4] > asof): continue
        P[(r[0], r[1])] = float(r[idx])
    Q = {e: {'s': s_, 'v': v} for (s_, e), v in P.items() if 80 <= _days(s_, e) <= 100}
    for _ in range(6):
        added = False
        for (s_, e), v in list(P.items()):
            if _days(s_, e) <= 100: continue
            for (s2, e2), v2 in list(P.items()):
                if s2 == s_ and e2 < e and 80 <= _days(e2, e) <= 100 and e not in Q:
                    Q[e] = {'s': (_dt.date.fromisoformat(e2) + _dt.timedelta(days=1)).isoformat(), 'v': v - v2}; added = True
            if e in Q:
                re_ = (_dt.date.fromisoformat(Q[e]['s']) - _dt.timedelta(days=1)).isoformat()
                if re_ > s_ and (s_, re_) not in P:
                    P[(s_, re_)] = v - Q[e]['v']; added = True
                    if 80 <= _days(s_, re_) <= 100 and re_ not in Q: Q[re_] = {'s': s_, 'v': v - Q[e]['v']}
        if not added: break
    return Q, P
def _contig(ends):
    """latest run of quarter ends with no gap (each 80-100 days after the previous)"""
    run = []
    for e in sorted(ends, reverse=True):
        if run and not (80 <= _days(e, run[-1]) <= 100): break
        run.append(e)
    return list(reversed(run))
def _plabel(s_, e):
    m = round(_days(s_, e) / 30.4)
    return ('quarter' if m == 3 else f'{m} months' if m != 12 else '12 months') + ' to ' + _qlabel(e)

def cash_block(B, fin, asof, fq):
    """Free cash flow history + last-12-month figure, and interest cover, from SEC; Yahoo only as a fallback for FCF."""
    cf = B.get('CFP', []); ip = B.get('INTP', [])
    O, OP = _qseries(cf, 2, asof); X, XP = _qseries(cf, 3, asof)
    ends = sorted(e for e in O if e in X)
    qs = [{'end': e, 'label': _qlabel(e), 'ocf': round(O[e]['v']), 'capex': round(X[e]['v']), 'fcf': round(O[e]['v'] - X[e]['v'])} for e in ends][-16:]
    run = _contig([q['end'] for q in qs]); byq = {q['end']: q for q in qs}
    for i, q in enumerate(qs):
        w = [x for x in qs[max(0, i - 3):i + 1]]
        q['ttm'] = sum(x['fcf'] for x in w) if len(w) == 4 and _contig([x['end'] for x in w]) == [x['end'] for x in w] else None
    ttm = byq[run[-1]]['ttm'] if run and len(run) >= 4 else None
    yago = None
    if ttm is not None:
        prev = [q for q in qs if 350 <= _days(q['end'], run[-1]) <= 380]
        yago = prev[-1]['ttm'] if prev else None
    periods = []
    for (s_, e), v in sorted(OP.items(), key=lambda kv: (kv[0][1], _days(*kv[0]))):
        if (s_, e) in XP and _days(s_, e) >= 80: periods.append({'start': s_, 'end': e, 'label': _plabel(s_, e), 'ocf': round(v), 'capex': round(XP[(s_, e)]), 'fcf': round(v - XP[(s_, e)])})
    revq = {q['end']: q['rev'] for q in fq if q.get('rev')}
    ttm_rev = sum(revq[e] for e in run[-4:]) if ttm is not None and all(e in revq for e in run[-4:]) else None
    if ttm is not None:
        fcf_v, fcf_lbl, fcf_src = ttm, f'last 12 months to {_qlabel(run[-1])}', 'SEC'
    elif periods:
        last = max(periods, key=lambda p: (p['end'], _days(p['start'], p['end'])))
        fcf_v, fcf_lbl, fcf_src = last['fcf'], last['label'], 'SEC'
    elif num(fin[2]) is not None:
        fcf_v, fcf_lbl, fcf_src = num(fin[2]) / 1e6, 'last 12 months (Yahoo)', 'Yahoo'
    else:
        fcf_v = fcf_lbl = fcf_src = None
    # interest cover
    OI, _ = _qseries(ip, 2, asof); IE, IEP = _qseries(ip, 3, asof)
    both = _contig([e for e in OI if e in IE])
    last_ie = max(IE) if IE else None
    cover = None; cov_txt = 'not reported'
    if both and (not OI or both[-1] == max(OI)):
        w = both[-4:]; a = sum(OI[e]['v'] for e in w); b = sum(IE[e]['v'] for e in w)
        lbl = f'last 12 months to {_qlabel(w[-1])}' if len(w) == 4 else f'quarter to {_qlabel(w[-1])}' if len(w) == 1 else f'{len(w)} quarters to {_qlabel(w[-1])}'
        if b > 0:
            cover = a / b
            cov_txt = f"{cover:.1f}x ({lbl}: operating {'profit' if a >= 0 else 'loss'} ${abs(a)/1e3:.1f}bn ÷ interest ${b/1e3:.2f}bn)" if abs(a) >= 1000 else f"{cover:.1f}x ({lbl}: operating {'profit' if a >= 0 else 'loss'} ${abs(a):.0f}m ÷ interest ${b:.0f}m)"
    elif last_ie:
        cov_txt = f'interest cost not reported separately since {_qlabel(last_ie)}'
    return {'quarters': qs, 'ttm': fcf_v, 'ttm_label': fcf_lbl, 'source': fcf_src, 'ttm_year_ago': yago, 'ttm_rev': ttm_rev,
            'periods': periods[-4:] if len(qs) < 2 else [], 'latest': run[-1] if run else (periods[-1]['end'] if periods else None),
            'cover': cover, 'cover_text': cov_txt}

    typical = move[3]; ratio = abs(move[0]) / typical if typical else None
    t5 = typical * math.sqrt(5) if typical else None; ratio5 = abs(move[1]) / t5 if t5 else None

    # ---- shareholders
    H = [{'name': h[0], 'date': h[1], 'shares': num(h[2]), 'chg': num(h[4])} for h in C.get('HOLDER', [])]
    latest = max((h['date'] for h in H if h['date']), key=lambda d: dt.datetime.strptime(d, '%m/%d/%Y'), default=None)
    def stale(h):
        if not h['date'] or not latest: return False
        return (dt.datetime.strptime(latest, '%m/%d/%Y') - dt.datetime.strptime(h['date'], '%m/%d/%Y')).days > 100   # older than the latest filing quarter = stale entity line
    def cls(c):
        if c is None: return 'n/a'
        return 'Adding' if c >= 5 else 'Maintaining' if c > -5 else 'Trimming' if c > -50 else 'Selling out'
    index_like = [s.lower() for s in S.get('index_holders', [])]
    is_index = lambda n: any(x in n.lower() for x in index_like)
    if first_13f: cls = lambda c: 'New holder'
    rows = [dict(h, disp=clean_name(h['name']), pct=(h['shares'] / (so * 1e6) * 100 if so else None), cls=cls(h['chg']), index=is_index(h['name'])) for h in H if not stale(h)]
    top5 = rows[:5]; active = [r for r in rows if not r['index']][:5]
    ins = [{'name': i[0].title(), 'role': i[1], 'buy_sh': num(i[2]), 'buy_val': num(i[3]), 'sell_sh': num(i[4]), 'sell_val': num(i[5])} for i in A.get('INSIDER', [])]

    # ---- closes for the chart
    cl = A['CLOSE1Y'][0]; closes = {'start': cl[0], 'end': cl[1], 'v': [int(x) / 10 for x in cl[2].split(',')]}

    data = {
        'meta': {'code': code, 'name': A['NAME'][0][0], 'exchange': S.get('exchange'), 'currency': A['NAME'][0][1], 'mcap': num(A['NAME'][0][2]),
                 'asof': asof, 'built': dt.datetime.utcnow().strftime('%Y-%m-%dT%H:%MZ'), 'status': S.get('status', 'staging'),
                 'version': S.get('version'), 'macro_src': macro_src, 'bench': S['benchmark']['name'], 'plan': 'Framework plan v0.5'},
        'price': {'px': px, 'd1': move[0], 'd5': move[1], 'm3': move[2], 'bench_d1': b1d, 'bench_d5': num(bench[2])},
        'call': call, 'zone': zn, 'composite': comp, 'coverage': round(cov, 2), 'buckets': buckets, 'bucket_w': BUCKET_W, 'bucket_live': bucket_live, 'driver_notes': driver_notes, 'lines': LINES,
        'direction': direction, 'confidence': conf, 'flags': flags, 'changers': changers,
        'indicators': list(I.values()), 'removed': S.get('removed_indicators', []),
        'new_listing': {'active': bool(nl_subs) or bool(dm2_until), 'listed': listed, 'days': ndays, 'subs': nl_subs, 'first_13f': first_13f, 'q13': q13.isoformat() if q13 else None, 'dm2_until': dm2_until},
        'stretch': {'v': stretch, 'flag': stretch is not None and stretch > 25, 'measured': "the average of every close since listing" if pt_short else "the stock's own 200-day average"},
        'peers': {'rows': [{'code': p[0], 'name': p[11] if len(p) > 11 else p[0], 'why': next((x['why'] for x in S['peers'] if x['code'] == p[0]), ''),
                            'fpe': num(p[4]), 'tpe': num(p[5]), 'peg': num(p[6]), 'ev': num(p[7]), 'revg': num(p[8]), 'gm': num(p[9]), 'r63': num(p[3])} for p in peers.values()],
                  'stock': {'fpe': fpe, 'tpe': tpe, 'peg': peg, 'ev': ev, 'revg': num(fin[0]), 'gm': num(fin[1])},
                  'median': {'fpe': m1, 'peg': m2, 'ev': m4}, 'excluded': [{'code': c, 'metric': m, 'value': x} for m, lst in (('P/E', ex1), ('PEG', ex2), ('EV/EBITDA', ex4)) for c, x in lst],
                  'plain': side.get('valuation_plain', [])},
        'move': {'d1': move[0], 'd5': move[1], 'typical': typical, 'ratio': ratio, 'ratio5': ratio5, 'unusual': (ratio or 0) >= MOVE_T or (ratio5 or 0) >= MOVE_T,
                 'beta': beta, 'beta_days': beta_n, 'beta_short': beta_short, 'parts': {'market': mkt, 'sector': sector, 'company': company}, **mv},
        'quality': {'checks': [{'name': c[0], 'flag': c[1], 'value': c[2], 'rule': c[3]} for c in checks], 'flags': nflags, 'cash': cash, 'info': info},
        'ev_calc': ev_calc,
        'shareholders': {'top5': top5, 'active': active, 'insiders': ins, 'asof': latest, 'shares_out_m': so, 'notes': side.get('shareholder_notes', []) + ([so_note] if so_note else [])},
        'earnings': side.get('earnings', {}), 'chains': side.get('chains', []), 'calendar': side.get('calendar', []),
        'macro': {k: v for k, v in macro.items()}, 'fed': {'meeting': fed[5] if fed[0] else None, 'meeting_note': 'the date the Polymarket market closes; the Fed usually announces its decision the day before', 'cut': cN, 'hike': hN, 'cut28': cT, 'hike28': hT},
        'lead': side.get('lead', ''), 'bullets': side.get('bullets', []), 'defects': side.get('defects', []) + [f"New-listing mode: {x['id']} uses a short-history version ({x['now']}) until trading day {x['end_day']} (about {x['end_est']}); short windows are noisier and include listing-related trading." for x in nl_subs] + ([f'LQ3 blank: the latest 13F quarter ({q13}) is the first since listing, so every holder shows as new.'] if first_13f else []) + ([f'DM2 left out until about {dm2_until}: the forecast 90 days ago predates analyst coverage' + (f'; after that it stays out while the base forecast is too small (${ago_e:.2f} vs ${now_e:.2f} today).' if dm2_small else '.')] if dm2_until else [f'DM2 left out: the forecast 90 days earlier (${ago_e:.2f}) is too small a base for a % change (${now_e:.2f} today).'] if dm2_small else []), 'sources': side.get('sources', []),
        'closes': closes, 'history': hist,
        'errors': [e for e in (A['ERRORS'][0][0], B['ERRORS'][0][0], C['ERRORS'][0][0]) if e],
    }
    for it in data['indicators']: it['explain'] = explain(it, S)
    data['mix'] = build_mix(S, D, side, (prev or {}).get('mix'), currency=data['meta']['currency'] or 'USD') if not commod else None
    if commod:
        # commodity variant (plan v0.6): company sections are repurposed; the template reads data.asset_type
        cm = S.get('commodity') or {}
        data['asset_type'] = 'commodity'

        data['commodity'] = dict(cfacts, asset=cm.get('asset', S.get('name')))
        data['demand_mix'] = side.get('demand_mix')

        fr = (cfacts.get('fred') or {}).get('DFII10')

        data['errors'] = [e for e in (A['ERRORS'][0][0], (G or {}).get('ERRORS', [['']])[0][0] if G else '') if e]
        if not G: data['errors'].append('commodity feeds (FRED, SPDR, CFTC) not collected: run stage G')
    data['changes'] = changes(prev, data)
    data['snapshots'] = snapshots(prev, data)
    macro_out = {'asof': asof, 'series': macro, 'fed': data['fed'], 'fed_slug': fed[0]}
    return data, macro_out
